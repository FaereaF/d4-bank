Мы возможно нашли не все дефекты, так что попросили нейронку исправить конфиг основываясь на найденных нами дефектах

 #HTTP: перенаправление всех запросов на HTTPS
server {
    listen 80;
    server_name bank.city.lab;

    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl;
    server_name bank.city.lab;

    ssl_certificate /certs/district.crt;
    ssl_certificate_key /certs/private/district.key;

    #Разрешены только TLS 1.2 и TLS 1.3
    ssl_protocols TLSv1.2 TLSv1.3;

    #Безопасные шифры TLS 1.2
    ssl_ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384;

    #Скрытие версии Nginx
    server_tokens off;

    #HSTS
    add_header Strict-Transport-Security "max-age=31536000" always;

    location / {
        proxy_pass https://api.bank.city.lab:8443;

        #Проверка сертификата API
        proxy_ssl_verify on;
        proxy_ssl_trusted_certificate /certs/ca.crt;
    }

    location /backup/ {
        alias /var/backups/bank/;
        autoindex on;
    }
}
